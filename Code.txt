import pandas as pd
from collections import defaultdict, deque
from datetime import timedelta


class AIOpsAlertEngine:
    def __init__(self, dependency_edges, time_window_minutes=15):
        """
        Initialize the AIOps engine with service architecture dependencies.
        :param dependency_edges: List of tuples representing service flows (e.g., [('DB_Service', 'Auth_Service')])
        :param time_window_minutes: Window size to cluster cascading alerts
        """
        self.time_window = timedelta(minutes=time_window_minutes)
        
        # Build a directed adjacency map: each service points to its dependencies.
        self.dep_graph = defaultdict(set)
        self.graph_nodes = set()
        for source, target in dependency_edges:
            self.dep_graph[source].add(target)
            self.graph_nodes.update((source, target))


        

    def _has_path(self, start, destination):
        """Return whether a directed path exists between two services."""
        pending = deque([start])
        visited = {start}
        while pending:
            service = pending.popleft()
            if service == destination:
                return True
            for neighbor in self.dep_graph.get(service, ()):
                if neighbor not in visited:
                    visited.add(neighbor)
                    pending.append(neighbor)
        return False


    def correlate_alerts(self, alerts_df):

        """
        Groups related alerts into a single incident-level view with probable origin highlighted.
        """
        alerts_df['timestamp'] = pd.to_datetime(alerts_df['timestamp'])
        alerts_df = alerts_df.sort_values('timestamp').reset_index(drop=True)
        
        incidents = []
        processed_alert_ids = set()
        
        incident_id = 1
        for idx, row in alerts_df.iterrows():
            if row['alert_id'] in processed_alert_ids:
                continue
                
            cluster = [row]
            processed_alert_ids.add(row['alert_id'])
            
            window_end = row['timestamp'] + self.time_window
            
            # Look ahead for cascading alerts within the time window
            for jdx, candidate in alerts_df.iloc[idx+1:].iterrows():
                if candidate['alert_id'] in processed_alert_ids:
                    continue
                
                if candidate['timestamp'] <= window_end:
                    # Check if services have a topological relationship or are the same service
                    serv_a = row['service']
                    serv_b = candidate['service']

                    is_connected = False
                    if serv_a == serv_b:
                        is_connected = True
                    elif self._has_path(serv_a, serv_b) or self._has_path(serv_b, serv_a):
                        is_connected = True
                        
                    if is_connected:
                        cluster.append(candidate)
                        processed_alert_ids.add(candidate['alert_id'])
            
            # Determine Probable Origin (Root Cause) using topological depth/incoming edges
            origin_service = self._find_root_cause(cluster)
            
            incidents.append({
                'incident_id': f"INC-{incident_id:03d}",
                'start_time': cluster[0]['timestamp'],
                'alert_count': len(cluster),
                'probable_origin': origin_service,
                'services_involved': list(set([c['service'] for c in cluster])),
                'explainability_trail': [f"[{c['timestamp']}] {c['service']} -> {c['message']}" for c in cluster],
                'raw_alerts': [c['alert_id'] for c in cluster]
            })
            incident_id += 1
            
        return pd.DataFrame(incidents)

    def _find_root_cause(self, cluster):
        """
        Identifies the most upstream service in the dependency graph within the cluster.
        """
        services = [c['service'] for c in cluster]
        service_set = set(services)
        if not service_set:
            return None

        # Find services with no incoming dependency edge within this cluster.
        incoming = {service: 0 for service in service_set}
        for source in service_set:
            for target in self.dep_graph.get(source, ()):
                if target in service_set:
                    incoming[target] += 1

        root_candidates = [service for service, count in incoming.items() if count == 0]
        return root_candidates[0] if root_candidates else services[0]


    def analyze_dead_weight_alerts(self, alerts_df, historical_incidents_df):
        """
        Identifies alerts that have never preceded a real historical incident.
        """
        # Get all alert types/signatures linked to historical incidents
        linked_alert_signatures = set(historical_incidents_df['alert_signature'].unique())
        
        # Analyze current alert stream signatures
        alerts_df['signature'] = alerts_df['service'] + ":" + alerts_df['alert_type']
        all_signatures = alerts_df['signature'].value_counts().reset_index()
        all_signatures.columns = ['signature', 'total_firings']
        
        # Find dead weights
        dead_weights = all_signatures[~all_signatures['signature'].isin(linked_alert_signatures)]
        
        recommendations = []
        for _, row in dead_weights.iterrows():
            recommendations.append({
                'alert_signature': row['signature'],
                'firings_count': row['total_firings'],
                'recommendation': 'DISABLE or INCREASE_THRESHOLD',
                'reason': 'Fired multiple times historically but never correlated with a verified real incident.'
            })
            
        return pd.DataFrame(recommendations)
